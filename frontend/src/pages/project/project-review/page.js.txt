var $ = function(i) { return document.getElementById(i); };
var api = window.api || window.__API__;
var auth = window.auth || window.__AUTH__;
var H = window.helpers || window.__DW_HELPERS__;
var esc = H && H.esc ? H.esc : function(s) {
  return String(s).replace(/[&<>"']/g, function(c) {
    return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
  });
};

var activeOrgId = null;
var activeProjectId = null;
var activeReviewId = null;
var currentProject = null;
var currentReview = null;
var currentDeliverable = null;
var allVersions = [];
var feedbackThreads = [];
var clientContactsMap = {};
var eligibleApproverNames = [];
var isManager = false;
var isViewer = false;
var activePinIndex = -1;
var pendingAnnotation = null;

// Sidebar toggle
var tog = $('tog');
if (tog) {
  tog.onclick = function() {
    var app = $('app');
    if (app) app.classList.toggle('min');
  };
}

function updateTabs() {
  var tabs = document.querySelector('.ptabs');
  if (tabs && activeProjectId) {
    tabs.querySelectorAll('a').forEach(function(a) {
      var href = a.getAttribute('href');
      if (href && href.indexOf('?') === -1 && href.indexOf('/workspace/') === -1) {
        a.setAttribute('href', href + '?project=' + encodeURIComponent(activeProjectId));
      }
    });
  }
}

function updateProjectHeader(p, clientName) {
  var top = document.querySelector('.top');
  if (!top || !p) return;
  var statusLabel = p.status ? p.status.charAt(0) + p.status.slice(1).toLowerCase() : 'Active';
  top.innerHTML = '<div><h1>' + esc(p.name) + '</h1><p>' + esc(clientName || 'Client') + ' · <span class="tag">' + esc(statusLabel) + '</span></p></div>';
}

function render() {
  if (!currentReview) return;

  var isClosed = currentReview.status !== 'OPEN';

  // 1. Status Banner
  var cl = $('cl');
  if (cl) {
    if (isClosed) {
      cl.hidden = false;
      if (currentReview.status === 'CANCELLED') {
        cl.textContent = 'This review was canceled. It’s read-only.';
      } else if (currentReview.status === 'SUPERSEDED') {
        cl.textContent = 'This review was replaced by a newer version. It’s read-only.';
      } else if (currentReview.status === 'APPROVED') {
        cl.textContent = 'This review was approved. It’s read-only.';
      } else if (currentReview.status === 'CHANGES_REQUESTED') {
        cl.textContent = 'Changes were requested on this review. It’s read-only.';
      } else {
        cl.textContent = 'This review is closed. It’s read-only.';
      }
    } else {
      cl.hidden = true;
    }
  }

  // 2. Key-Value Row
  var kd = $('kd');
  if (kd) kd.textContent = currentDeliverable ? currentDeliverable.title : 'Deliverable';

  var kvn = $('kvn');
  if (kvn) {
    var vNum = currentReview.version ? 'v' + currentReview.version.version_number : '—';
    var vDate = currentReview.created_at ? (H.formatDate ? H.formatDate(currentReview.created_at) : currentReview.created_at.slice(0, 10)) : '';
    kvn.textContent = vNum + (vDate ? ' · ' + vDate : '');
  }

  var cs = $('cs');
  if (cs) {
    if (currentReview.status === 'OPEN') cs.textContent = 'Open';
    else if (currentReview.status === 'APPROVED') cs.textContent = 'Approved';
    else if (currentReview.status === 'CHANGES_REQUESTED') cs.textContent = 'Changes requested';
    else if (currentReview.status === 'CANCELLED') cs.textContent = 'Canceled';
    else if (currentReview.status === 'SUPERSEDED') cs.textContent = 'Replaced';
    else cs.textContent = currentReview.status;
  }

  var kdue = $('kdue');
  if (kdue) {
    kdue.textContent = currentReview.due_at ? (H.formatDate ? H.formatDate(currentReview.due_at) : currentReview.due_at.slice(0, 10)) : 'No due date';
  }

  // 3. Media Preview & Header Bar
  var pfn = $('pfn');
  var pvn = $('pvn');
  if (pfn) pfn.textContent = (currentDeliverable ? currentDeliverable.title : 'asset') + ' (v' + (currentReview.version ? currentReview.version.version_number : 1) + ')';
  if (pvn) pvn.textContent = 'v' + (currentReview.version ? currentReview.version.version_number : 1);

  var pvw = $('pvw');
  var np = $('np');
  var pvimg = $('pvimg');
  var pins = $('pins');

  if (currentReview.previews && currentReview.previews.length > 0) {
    if (pvw) pvw.hidden = false;
    if (np) np.hidden = true;
    if (pvimg) {
      pvimg.src = '/api/v1/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/reviews/' + activeReviewId + '/preview/1';
      pvimg.style.display = 'block';
    }
  } else {
    // If no previews, show placeholder/empty
    if (pvw) pvw.hidden = true;
    if (np) np.hidden = false;
  }

  // 4. Render Pins on preview
  if (pins) {
    pins.innerHTML = feedbackThreads.map(function(t, i) {
      if (!t.annotation || t.annotation.x == null || t.annotation.y == null) return '';
      var isOn = activePinIndex === i;
      var xPct = (t.annotation.x * 100).toFixed(1);
      var yPct = (t.annotation.y * 100).toFixed(1);
      return '<button class="pin' + (isOn ? ' on' : '') + '" style="position:absolute;left:' + xPct + '%;top:' + yPct + '%" data-p="' + i + '" aria-label="Comment ' + (i + 1) + '">' + (i + 1) + '</button>';
    }).join('');
  }

  // 5. Decision Log & Approver Info
  var dlog = $('dlog');
  if (dlog) {
    if (currentReview.decision) {
      var d = currentReview.decision;
      var dDate = d.created_at ? (H.formatDate ? H.formatDate(d.created_at) : d.created_at.slice(0, 10)) : '';
      var dText = d.decision === 'APPROVED' ? 'Approved' : 'Requested changes: ' + (d.reason || '');
      dlog.innerHTML = '<li><time>' + esc(dDate) + '</time>v' + (currentReview.version ? currentReview.version.version_number : '') + ' · ' + esc(dText) + '</li>';
    } else {
      dlog.innerHTML = '<li><time>Pending</time>No formal decision has been submitted yet.</li>';
    }
  }

  var cd = $('cd');
  if (cd) {
    if (eligibleApproverNames.length > 0) {
      cd.textContent = 'Who can decide: ' + eligibleApproverNames.join(', ');
    } else {
      cd.textContent = 'Who can decide: Client Approvers';
    }
  }

  // 6. Comments List
  var cm = $('cm');
  if (cm) {
    if (feedbackThreads.length === 0) {
      cm.innerHTML = '<p class="empty" style="padding:16px 0"><b>No comments yet</b>Click the preview to leave a pinpoint note, or add a note below.</p>';
    } else {
      cm.innerHTML = feedbackThreads.map(function(t, i) {
        var isOn = activePinIndex === i;
        var author = t.actor_name || (t.capacity === 'CLIENT' ? 'Client reviewer' : 'Team member');
        var dateStr = t.created_at ? (H.formatDate ? H.formatDate(t.created_at) : t.created_at.slice(0, 10)) : '';
        var isResolved = t.resolution === 'RESOLVED';
        var hasTask = !!t.task_id;

        var statusTags = '';
        if (isResolved) statusTags += ' <span class="st ok">Resolved</span>';
        if (hasTask) statusTags += ' <span class="st">Task created</span>';

        var repliesHtml = (t.replies || []).map(function(r) {
          var rAuthor = r.actor_name || (r.capacity === 'CLIENT' ? 'Client' : 'Team');
          return '<div class="rp"><b>' + esc(rAuthor) + ':</b> ' + esc(r.body) + '</div>';
        }).join('');

        var actionsHtml = '';
        if (!isClosed && !isViewer) {
          var resBtn = '<button class="btn sm" data-r="' + i + '">' + (isResolved ? 'Reopen' : 'Resolve') + '</button>';
          var taskBtn = hasTask
            ? '<button class="btn sm" disabled title="Task already created">Task created</button>'
            : '<button class="btn sm" data-t="' + i + '">Turn into task</button>';
          actionsHtml = '<div class="row2" style="margin-top:8px">' + resBtn + (isManager ? taskBtn : '') + '</div>';
        }

        return '<div class="cmt' + (isOn ? ' on' : '') + '" data-c="' + i + '" style="margin-bottom:12px;padding:12px;background:var(--card,#fff);border-radius:12px;border:1px solid var(--border,#e0e0e0)">' +
          '<b style="font-weight:500">' + (i + 1) + '. ' + esc(author) + '</b> <small style="color:var(--muted)">' + esc(dateStr) + '</small>' + statusTags +
          '<div style="margin-top:6px">' + esc(t.body) + '</div>' +
          repliesHtml +
          actionsHtml +
          '</div>';
      }).join('');
    }
  }

  // 7. Input Controls State
  var an = $('an');
  var nn = $('nn');
  var rp = $('rp');
  var ap = $('ap');

  if (an) an.disabled = isClosed || isViewer;
  if (nn) nn.disabled = isClosed || isViewer;
  if (rp) {
    rp.disabled = isClosed || !isManager;
    rp.style.display = isManager ? '' : 'none';
  }
  if (ap) {
    // Internal team never approves; only client approvers in the portal do
    ap.disabled = true;
    ap.title = 'Internal staff can’t approve. Only the client’s decision-makers can.';
  }

  wireInteractions();
}

function wireInteractions() {
  // Pin clicks
  document.querySelectorAll('[data-p]').forEach(function(b) {
    b.onclick = function() {
      activePinIndex = Number(b.getAttribute('data-p'));
      render();
    };
  });

  // Comment card clicks
  document.querySelectorAll('[data-c]').forEach(function(d) {
    d.onclick = function(e) {
      if (e.target.closest('button')) return;
      activePinIndex = Number(d.getAttribute('data-c'));
      render();
    };
  });

  // Resolve / Reopen
  document.querySelectorAll('[data-r]').forEach(function(b) {
    b.onclick = async function() {
      var i = Number(b.getAttribute('data-r'));
      var t = feedbackThreads[i];
      var isResolved = t.resolution === 'RESOLVED';
      var newState = isResolved ? 'REOPENED' : 'RESOLVED';

      try {
        await api.post(
          '/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/reviews/' + activeReviewId + '/feedback/' + t.id + '/resolution',
          { state: newState, note: 'Updated via review workspace' }
        );
        var ce = $('ce');
        if (ce) ce.hidden = true;
        await loadData();
      } catch (err) {
        var ce = $('ce');
        if (ce) {
          ce.textContent = err.message || 'Failed to update feedback resolution.';
          ce.hidden = false;
        }
      }
    };
  });

  // Turn into task
  document.querySelectorAll('[data-t]').forEach(function(b) {
    b.onclick = async function() {
      var i = Number(b.getAttribute('data-t'));
      var t = feedbackThreads[i];
      if (t.task_id) {
        var ce = $('ce');
        if (ce) {
          ce.textContent = 'This comment is already a task.';
          ce.hidden = false;
        }
        return;
      }

      try {
        await api.post(
          '/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/reviews/' + activeReviewId + '/feedback/' + t.id + '/task',
          {
            title: 'Fix: ' + t.body.slice(0, 50),
            description: t.body,
            priority: 'NORMAL',
            status: 'TODO',
            due_at: null,
            milestone_id: null,
            assignee_id: null,
            position: 0
          }
        );
        var ce = $('ce');
        if (ce) ce.hidden = true;
        var ok = $('ok');
        if (ok) {
          ok.textContent = 'Internal task created. The client won’t see it.';
          ok.hidden = false;
        }
        await loadData();
      } catch (err) {
        var ce = $('ce');
        if (ce) {
          ce.textContent = err.message || 'Failed to turn feedback into task.';
          ce.hidden = false;
        }
      }
    };
  });

  // Preview click for pin annotation
  var pvprev = $('pvprev');
  if (pvprev) {
    pvprev.onclick = function(e) {
      if (e.target.closest('.pin')) return;
      var rect = pvprev.getBoundingClientRect();
      var clickX = e.clientX - rect.left;
      var clickY = e.clientY - rect.top;
      var xCoord = Math.max(0, Math.min(1, clickX / rect.width));
      var yCoord = Math.max(0, Math.min(1, clickY / rect.height));

      pendingAnnotation = {
        type: 'IMAGE_PIN',
        x: Math.round(xCoord * 1000) / 1000,
        y: Math.round(yCoord * 1000) / 1000
      };

      var nn = $('nn');
      if (nn) {
        nn.focus();
        nn.placeholder = 'Add note at pin (' + (pendingAnnotation.x * 100).toFixed(1) + '%, ' + (pendingAnnotation.y * 100).toFixed(1) + '%)…';
      }
    };
  }
}

async function loadData() {
  if (!activeOrgId || !activeProjectId || !activeReviewId) return;

  // 1. Fetch Review Details
  try {
    currentReview = await api.get('/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/reviews/' + activeReviewId);
  } catch (err) {
    console.error('Failed to load review:', err);
    if (H && H.handlePageAccessError && H.handlePageAccessError(err, 'app')) return;
    return;
  }

  // 2. Fetch Deliverable Details
  if (currentReview.deliverable_id) {
    try {
      currentDeliverable = await api.get('/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/deliverables/' + currentReview.deliverable_id);
    } catch (_) {}
  }

  // 3. Fetch Versions of Deliverable (for replacement dialog)
  try {
    var vRes = await api.get('/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/versions?limit=100');
    var vItems = vRes.items || [];
    allVersions = vItems.filter(function(v) {
      return v.deliverable_id === currentReview.deliverable_id;
    });
  } catch (_) {}

  // 4. Fetch Client Grants & Contacts for "Who can decide"
  try {
    var gRes = await api.get('/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/grants');
    var grants = Array.isArray(gRes) ? gRes : (gRes.items || []);
    var approverGrants = grants.filter(function(g) { return g.role === 'APPROVER' && g.status === 'ACTIVE'; });

    var contactsRes = await api.get('/organizations/' + activeOrgId + '/clients/' + currentProject.client_id + '/contacts');
    var contacts = Array.isArray(contactsRes) ? contactsRes : (contactsRes.items || []);

    eligibleApproverNames = [];
    approverGrants.forEach(function(g) {
      var c = contacts.find(function(item) { return item.id === g.contact_id; });
      if (c && c.name) {
        eligibleApproverNames.push(c.name);
      }
    });
  } catch (_) {}

  // 5. Fetch Feedback Threads
  try {
    var fRes = await api.get('/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/reviews/' + activeReviewId + '/feedback?limit=100');
    feedbackThreads = fRes.items || [];
  } catch (err) {
    console.error('Failed to load feedback:', err);
  }

  render();
}

async function init() {
  var session = await auth.init();
  if (!session || !session.authenticated) return;

  if (H && H.loadHeader) {
    await H.loadHeader();
  }

  activeOrgId = auth.getActiveOrgId ? auth.getActiveOrgId() : null;
  if (!activeOrgId && session.principals && session.principals.length > 0) {
    activeOrgId = session.principals[0].organizationId || session.principals[0].organization_id;
  }

  var params = new URLSearchParams(window.location.search);
  activeProjectId = params.get('project') || params.get('id');
  activeReviewId = params.get('review');

  if (!activeProjectId) {
    var pRes = await api.get('/organizations/' + activeOrgId + '/projects?limit=1');
    if (pRes && pRes.items && pRes.items.length) {
      activeProjectId = pRes.items[0].id;
    }
  }

  if (!activeProjectId) return;
  updateTabs();

  // Load project details
  try {
    currentProject = await api.get('/organizations/' + activeOrgId + '/projects/' + activeProjectId);
    var clientName = 'Client';
    if (currentProject && currentProject.client_id) {
      try {
        var clientObj = await api.get('/organizations/' + activeOrgId + '/clients/' + currentProject.client_id);
        if (clientObj) clientName = clientObj.name;
      } catch (_) {}
    }
    updateProjectHeader(currentProject, clientName);
  } catch (err) {
    if (H && H.handlePageAccessError && H.handlePageAccessError(err, 'app')) return;
  }

  // Load user role
  try {
    var membersRes = await api.get('/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/members?limit=100');
    var members = membersRes.items || [];
    var principalId = session.principals ? session.principals[0].id : null;
    var myMembership = members.find(function(m) { return m.membership_id === principalId; });
    var role = myMembership ? myMembership.role : null;
    var orgRole = session.principals ? session.principals[0].role : null;

    isManager = role === 'MANAGER' || orgRole === 'OWNER' || orgRole === 'ADMIN';
    isViewer = role === 'VIEWER';
  } catch (_) {}

  // If review ID not given in URL, look up the latest open review in project
  if (!activeReviewId) {
    try {
      var rList = await api.get('/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/reviews?limit=10');
      var rItems = rList.items || [];
      var openR = rItems.find(function(r) { return r.status === 'OPEN'; });
      if (openR) activeReviewId = openR.id;
      else if (rItems.length > 0) activeReviewId = rItems[0].id;
    } catch (_) {}
  }

  // Wire Add Note button
  var an = $('an');
  if (an) {
    an.onclick = async function() {
      var nn = $('nn');
      var text = nn ? nn.value.trim() : '';
      if (!text) return;

      var ce = $('ce');
      var ok = $('ok');
      if (ce) ce.hidden = true;

      try {
        await api.post(
          '/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/reviews/' + activeReviewId + '/feedback',
          {
            version_id: currentReview.version_id,
            digest: currentReview.version.digest,
            visibility: 'CLIENT',
            body: text,
            annotation: pendingAnnotation || null
          }
        );

        if (nn) {
          nn.value = '';
          nn.placeholder = '';
        }
        pendingAnnotation = null;
        await loadData();
      } catch (err) {
        if (ce) {
          ce.textContent = err.message || 'Failed to add feedback.';
          ce.hidden = false;
        }
      }
    };
  }

  // Wire Cancel / Replace Review Dialog
  var rp = $('rp');
  var dg = $('dg');
  var dgf = $('dgf');
  var ac = $('ac');
  var rvf = $('rvf');
  var rsel = $('rsel');
  var dgerr = $('dgerr');

  if (rp && dg) {
    rp.onclick = function() {
      if (dgerr) dgerr.hidden = true;
      if (ac) ac.value = 'cancel';
      if (rvf) rvf.hidden = true;

      // Populate newer versions in rsel
      if (rsel && currentReview && currentReview.version) {
        var curVerNum = currentReview.version.version_number;
        var newerVersions = allVersions.filter(function(v) {
          return v.version_number > curVerNum;
        });
        rsel.innerHTML = newerVersions.map(function(v) {
          return '<option value="' + esc(v.id) + '">v' + esc(v.version_number) + '</option>';
        }).join('');
      }

      dg.showModal();
    };
  }

  if (ac && rvf) {
    ac.onchange = function() {
      rvf.hidden = ac.value !== 'replace';
    };
  }

  if (dgf && dg) {
    dgf.onsubmit = async function(e) {
      e.preventDefault();
      var actionVal = ac ? ac.value : 'cancel';

      try {
        if (actionVal === 'cancel') {
          await api.post(
            '/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/reviews/' + activeReviewId + '/cancel',
            {},
            { headers: { 'If-Match': '"' + currentReview.revision + '"' } }
          );
        } else if (actionVal === 'replace') {
          var targetVersionId = rsel ? rsel.value : null;
          var targetVersion = allVersions.find(function(v) { return v.id === targetVersionId; });
          if (!targetVersion) {
            throw new Error('Select a newer version to replace the review.');
          }

          var approverGrants = (currentReview.participants || []).map(function(p) { return p.contact_id; });

          var replaceRes = await api.post(
            '/organizations/' + activeOrgId + '/projects/' + activeProjectId + '/reviews/' + activeReviewId + '/replace',
            {
              version_id: targetVersion.id,
              digest: targetVersion.digest,
              approver_ids: approverGrants,
              due_at: currentReview.due_at || new Date(Date.now() + 86400000 * 3).toISOString()
            },
            { headers: { 'If-Match': '"' + currentReview.revision + '"' } }
          );

          if (replaceRes && replaceRes.id) {
            activeReviewId = replaceRes.id;
          }
        }

        dg.close();
        await loadData();
      } catch (err) {
        if (dgerr) {
          dgerr.textContent = err.message || 'Action failed.';
          dgerr.hidden = false;
        }
      }
    };
  }

  // Wire Cancel buttons on all dialogs
  document.querySelectorAll('[data-x]').forEach(function(b) {
    b.onclick = function() {
      var d = b.closest('dialog');
      if (d) d.close();
    };
  });

  await loadData();
}

if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', init);
} else {
  init();
}
